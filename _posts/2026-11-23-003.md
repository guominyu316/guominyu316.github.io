  // 原始得分率（卷面分）

  // 详细数据字典（用于Tooltip展示分数和名次）
  const detailedData = {
    '语文(150)': { total: 96, raw: 96, full: 150, rank: { class: 48, school: 733, all: 10050 } },
    '数学(150)': { total: 76, raw: 76, full: 150, rank: { class: 16, school: 238, all: 3593 } },
    '英语(150)': { total: 116, raw: 116, full: 150, rank: { class: 7, school: 112, all: 1186 } },
    '物理(100)': { total: 66, raw: 66, full: 100, rank: { class: 21, school: 301, all: 1806 } },
    '化学(100)': { total: 88, raw: 63, full: 100, rank: { class: 4, school: 75, all: 1090 } },
    '生物(100)': { total: 95, raw: 88, full: 100, rank: { class: 1, school: 9, all: 61 } }
  };

  new Chart(document.getElementById('radar1'), {
    type: 'radar',
    data: {
      labels: subjects,
      datasets: [
        {
          label: '计入总分得分率 (化生为赋分)',
          data: totalScoreRate,
          fill: true,
          backgroundColor: 'rgba(58, 110, 165, 0.2)',
          borderColor: 'rgba(58, 110, 165, 1)',
          pointBackgroundColor: 'rgba(58, 110, 165, 1)',
          pointBorderColor: '#fff',
          pointRadius: 4,
          pointHoverRadius: 6
        },
        {
          label: '原始得分率 (卷面)',
          data: rawScoreRate,
          fill: false,
          borderColor: 'rgba(255, 107, 107, 1)',
          pointBackgroundColor: 'rgba(255, 107, 107, 1)',
          pointBorderColor: '#fff',
          borderDash: [5, 5],
          pointRadius: 4,
          pointHoverRadius: 6
        }
      ]

            label: function(context) {

              const info = detailedData[sub];
              const totalPct = (info.total / info.full * 100).toFixed(1);
              const rawPct = (info.raw / info.full * 100).toFixed(1);
              
              let lines = [];
              
              // 只有化学和生物才展示赋分对比，其他展示原始分

                lines.push(`赋分 / 原始: ${info.total}分 / ${info.raw}分`);
                lines.push(`得分率: ${totalPct}% / ${rawPct}%`);
              } else {
                lines.push(`计入总分(原始分): ${info.total}分`);
                lines.push(`得分率: ${totalPct}%`);
              }
              lines.push(`班名次: 第 ${info.rank.class} 名`);
              lines.push(`校名次: 第 ${info.rank.school} 名`);
              lines.push(`全体名次: 第 ${info.rank.all} 名`);
              
              return lines;
            }
          }
        },
        legend: {
          position: 'top',
        }
      },
      scales: {
        r: {
          beginAtZero: true,
          max: 100,
          ticks: { stepSize: 20 },
          pointLabels: {
            font: { size: 13 }
          }
        }
      }
    }
  });
});
</script>

## 看图结论与学情分析

### 1. 绝对优势科目
*   **生物**：赋分 **95分 (A)**，原始分 88分。**班级第1，校排第9，全体第61名**。这是你最核心的拉分项。
*   **化学**：赋分 **88分 (A)**，原始分 63分。虽然卷面分不高，但**班级第4，校排第75**，说明排名非常靠前，赋分暴涨 **+25分**。

### 2. 中坚力量
*   **英语**：116分（得分率 77.3%），班级第7，校排112。表现稳定。
*   **物理**：66分（得分率 66%），班级第21，校排301。中规中矩，有提升空间。
*   **数学**：76分（得分率 50.7%），但班级排名 **第16**。这说明试卷难度极大，你的名次其实不差，但绝对分数仍需重点突破。

### 3. 严重短板
*   **语文**：96分（得分率 64%），**班级排名 48（倒数），全体排名 10050**。这是目前最拖后腿的科目，亟需重视。

### 💡 提分建议排序
1.  **语文**：优先挽救。排名垫底，哪怕只提升10分，总排名也会有质的飞跃。
2.  **数学**：试卷难的情况下你排名16，说明基础可以。需要攻克中高难度题。
3.  **化学**：虽然赋分高，但原始分只有63。新高考赋分受试卷难度影响大，必须把卷面底子打牢。
4.  **生物**：保持现有优势即可。
